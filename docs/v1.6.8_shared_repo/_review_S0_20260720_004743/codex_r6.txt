Use this skill when a task benefits from the context-mode MCP server installed from `mksglu/context-mode`.

- Routes Codex tool use through context-mode hooks.

export function validateIndex(text: string): ValidateResult<RegistryIndex> {
  const parsed = safeParseJson(text);
  if (!parsed.ok) return { ok: false, error: parsed.error };

  if (!parsed.ok) return MetaSchema.parse({});
  const res = MetaSchema.safeParse(parsed.value);
  return res.success ? res.data : MetaSchema.parse({});
}

// 안전 read → tmp 채널 파일 기록. 실패(거부)면 skipped 기록.
async function materialize(
  root: string,
  segs: string[],
  kind: Parameters<typeof readRegFile>[2],
  tmp: string,
  dest: string[],
  skipped: RegistryEval["skipped"],
): Promise<boolean> {
  const r = await readRegFile(root, segs, kind);
  if (!r.ok) {
    skipped.push({ path: segs.join("/"), reason: r.error });
    return false;
  }
  const full = join(tmp, ...dest);
  await mkdir(join(full, ".."), { recursive: true });
  await writeFile(full, r.text, "utf8");
  return true;
}

// 문자열-인지 raw 중첩 깊이 스캐너(JSON). "..." 내부의 괄호는 무시(false-reject 최소화).
//   파싱 前 호출 → 깊은 중첩을 JSON.parse(스택 오버플로 위험) 전에 fail-closed 거부.
export function rawJsonDepth(text: string): number {
  let depth = 0;
  let max = 0;
  let inStr = false;
  let esc = false;
  for (let i = 0; i < text.length; i++) {
    const c = text[i];
    if (inStr) {
      if (esc) esc = false;
      else if (c === "\\") esc = true;
      else if (c === '"') inStr = false;
      continue;
    }
    if (c === '"') inStr = true;
    else if (c === "{" || c === "[") {
      depth++;
      if (depth > max) max = depth;
    } else if (c === "}" || c === "]") {
      if (depth > 0) depth--;
    }
  }
  return max;
}

// 파싱 결과 객체 깊이(BFS·비재귀 — 스택 안전·노드 예산 제한). cap 초과 즉시 true.
// nodeBudget = BFS 순회 노드 상한(깊이가 아닌 노드폭발 DoS 보조가드). 기본은 넉넉히 —
//   JSON.parse 가 이미 maxBytes 로 트리 크기를 제한하므로 정상 얕고-넓은 파일(대형 카탈로그)이
//   오탐 거부되지 않게 한다(agy MED: 200k 로 16MB index 오탐 거부 수정). 호출측이 크기 비례로 지정.
export function objectDepthExceeds(value: unknown, cap: number, nodeBudget = 10_000_000): boolean {
  let budget = nodeBudget;
  let frontier: unknown[] = [value];
  let depth = 0;
  while (frontier.length > 0) {
    if (depth > cap) return true;
    const next: unknown[] = [];
    for (const node of frontier) {
      if (--budget <= 0) return true; // 노드 폭발도 DoS → 거부
      if (node && typeof node === "object") {
        for (const v of Object.values(node as Record<string, unknown>)) next.push(v);
      }
    }
    frontier = next;
    depth++;
  }
  return false;
}

export type ParseResult<T = unknown> = { ok: true; value: T } | { ok: false; error: string };

// YAML safe-load: `!!` 코드/커스텀 태그 사전 거부(방어적·eemeli 는 함수 실행 안 하나 명시 차단)
//   + alias 확장 상한(maxAliasCount·billion-laughs) + 깊이 상한.
const SAFE_YAML_TAG = /(^|[\s,{[])!!(?!str\b|int\b|float\b|bool\b|null\b|map\b|seq\b|binary\b|timestamp\b)/m;

  if (SAFE_YAML_TAG.test(text)) return { ok: false, error: "unsafe-yaml-tag" };

  } catch (e) {

// JSON safe-parse: 크기 상한 → raw 깊이 상한(스택 방어) → JSON.parse → 결과 깊이 재확인.
export function safeParseJson<T = unknown>(text: string, maxBytes = REG_MAX_INDEX): ParseResult<T> {
  if (Buffer.byteLength(text, "utf8") > maxBytes) return { ok: false, error: "too-large" };
  if (rawJsonDepth(text) > REG_MAX_STRUCT_DEPTH) return { ok: false, error: "json-too-deep" };
  let value: unknown;
  try {
    value = JSON.parse(text);
  } catch (e) {
    return { ok: false, error: "json-parse:" + (e instanceof Error ? e.message : String(e)) };
  }
  // 노드예산은 입력 크기 비례(정상 대형 카탈로그 오탐 방지·트리는 이미 maxBytes 로 제한됨).
  if (objectDepthExceeds(value, REG_MAX_STRUCT_DEPTH, Math.max(10_000_000, Buffer.byteLength(text, "utf8")))) {
    return { ok: false, error: "json-too-deep" };
  }
  return { ok: true, value: value as T };
}

// TOML safe-parse: 크기 상한 → @iarna strict parse(코드 실행 없음) → 깊이 상한.
export function safeParseToml(text: string, maxBytes = REG_MAX_ARTIFACT): ParseResult {
  if (Buffer.byteLength(text, "utf8") > maxBytes) return { ok: false, error: "too-large" };

    value = TOML.parse(text);

}

export async function readRegFile(root: string, segs: string[], kind: RegKind): Promise<ReadResult> {
  if (segs.length === 0 || segs.some((s) => !isSafeSegment(s))) return { ok: false, error: "invalid-path" };
